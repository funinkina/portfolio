{{- /* Resizes a local asset or a remote URL to .width, emitting webp + fallback. Unprocessable formats (gif, svg) are emitted as-is; unreachable remote images render nothing rather than a broken . */ -}} {{- $img := partial "image-resource.html" .path -}} {{- $ext := path.Ext .path | lower -}} {{- if and $img (not (in (slice ".gif" ".svg") $ext)) -}} {{- $webp := $img.Resize (printf "%dx webp" .width) -}} {{- $fallback := $img.Resize (printf "%dx" .width) -}} {{- else if $img -}} {{- else if not (hasPrefix .path "http") -}} {{- end -}}
